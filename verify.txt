echo "--- automatic branch check ---"
if git branch -vv | grep -E "ahead [0-9]+" ; then
  echo "PROBLEM: branch(es) above have UNPUSHED commits - message before doing anything"
else
  echo "OK: no unpushed commits on any branch"
fi
for b in main arpa shadman sunanda2; do
  if git rev-parse --verify -q refs/heads/$b >/dev/null; then
    if [ "$(git rev-parse $b)" = "$(git rev-parse origin/$b)" ]; then
      echo "OK: $b matches origin/$b"
    else
      echo "PROBLEM: $b does NOT match origin/$b - run the repair steps for this branch"
    fi
  else
    echo "SKIP: $b not on this machine (nothing to do)"
  fi
done
